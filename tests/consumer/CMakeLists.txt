cmake_minimum_required(VERSION 3.16)
project(SuperJupiterSignalFilterConsumer LANGUAGES CXX)
if(DEFINED SUPERJUPITER_SIGNAL_FILTER_SOURCE_DIR)
  add_subdirectory("${SUPERJUPITER_SIGNAL_FILTER_SOURCE_DIR}" signal-filter)
  if(TARGET superjupiter-signal-filter-tests OR TARGET superjupiter-signal-filter-replay)
    message(FATAL_ERROR "A consumer unexpectedly builds library developer tools")
  endif()
else()
  find_package(SuperJupiterSignalFilter 0.1 CONFIG REQUIRED)
endif()
add_executable(superjupiter-signal-filter-consumer main.cpp)
target_link_libraries(superjupiter-signal-filter-consumer PRIVATE SuperJupiter::SignalFilter)
